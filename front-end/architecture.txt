Perfect bro, here’s a clean example of a getFarmerCrops service inside /features/farmers/services/farmers.js. This service is just responsible for making API calls to your backend and returning data. Nothing else.

// /features/farmers/services/farmers.js

import axios from 'axios';

// Base URL for your backend API
const API_BASE = process.env.REACT_APP_API_URL || 'http://localhost:4000/api';

/**
 * Fetch crops for a specific farmer by ID
 * @param {string} farmerId
 * @returns {Promise<Array>} list of crops
 */
export const getFarmerCrops = async (farmerId) => {
  try {
    const response = await axios.get(`${API_BASE}/farmers/${farmerId}/crops`);
    return response.data; // the API should return an array of crops
  } catch (error) {
    console.error('Error fetching farmer crops:', error);
    throw error; // allow hooks or pages to handle the error
  }
};
✅ How it’s used with a hook
// /features/farmers/hooks/useCrops.js
import { useState, useEffect } from 'react';
import { getFarmerCrops } from '../services/farmers';

export const useCrops = (farmerId) => {
  const [crops, setCrops] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    if (!farmerId) return;

    setLoading(true);
    getFarmerCrops(farmerId)
      .then((data) => setCrops(data))
      .catch((err) => setError(err))
      .finally(() => setLoading(false));
  }, [farmerId]);

  return { crops, loading, error };
};

Then your page just uses it:

// /features/farmers/pages/FarmerDashboard.js
import React from 'react';
import { useCrops } from '../hooks/useCrops';
import CropCard from '../components/CropCard';

const FarmerDashboard = ({ farmerId }) => {
  const { crops, loading, error } = useCrops(farmerId);

  if (loading) return <div>Loading...</div>;
  if (error) return <div>Error fetching crops!</div>;

  return (
    <div>
      {crops.map((crop) => (
        <CropCard key={crop.id} crop={crop} />
      ))}
    </div>
  );
};

export default FarmerDashboard;

This shows feature-based architecture in action:

Service → hook → page → component
Page doesn’t call API directly
Components are dumb and only receive props

This keeps your frontend scalable and clean.
